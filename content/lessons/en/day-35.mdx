## Intro

<En>
You have learned all the building blocks of professional Playwright automation: locators, actions, waiting, screenshots, POM, fixtures, parametrize, debugging, and CI. Now it is time to put them all together.

Your mini project is to build a **complete, professional-grade test suite** with at least 10 tests against `https://practicesoftwaretesting.com`. The suite must use Page Object Model, pytest fixtures, parametrize where appropriate, and produce meaningful failure output (screenshots on failure). This is the kind of suite you would show in a job interview or add to your GitHub portfolio.
</En>
<Vi>
Bạn đã học tất cả các khối xây dựng của tự động hoá Playwright chuyên nghiệp: locator, action, chờ đợi, screenshot, POM, fixture, parametrize, debug và CI. Bây giờ là lúc kết hợp tất cả lại.

Dự án nhỏ của bạn là xây dựng một **bộ test hoàn chỉnh, chuyên nghiệp** với ít nhất 10 test cho `https://practicesoftwaretesting.com`. Bộ test phải dùng Page Object Model, pytest fixture, parametrize khi thích hợp và tạo ra output thất bại có ý nghĩa (screenshot khi thất bại). **Quality checklist for a professional suite:**
- [ ] POM — no raw locators in test files
- [ ] Fixtures — login is in a shared fixture, not copied
- [ ] Parametrize — repeated logic uses `@pytest.mark.parametrize`
- [ ] Clear test names — readable, describe what is being tested
- [ ] Assertions with messages — `assert x > 0, "Expected at least one product"`
- [ ] No `time.sleep()` — use Playwright waits
- [ ] CI workflow — tests run on push to main
- [ ] Failure artifacts — screenshots and traces on failure

```python
# pages/product_page.py
from __future__ import annotations
from playwright.sync_api import Page, Locator

class ProductPage:
    """
    Page Object for individual product pages at:
    https://practicesoftwaretesting.com/product/{id}
    """

    def __init__(self, page: Page) -> None:
        self.page = page

        # ── Locators ────────────────────────────────────────────────
        self.product_title: Locator = page.locator(".product-title, h1")
        self.product_price: Locator = page.locator(".product-price, .price")
        self.add_to_cart_btn: Locator = page.get_by_role("button", name="Add to cart")
        self.quantity_input: Locator = page.locator("input[type='number']")
        self.success_toast: Locator = page.locator(".toast-success, .alert-success")

    # ── Queries ─────────────────────────────────────────────────────
    def get_title(self) -> str:
        return self.product_title.text_content() or ""

    def get_price(self) -> str:
        return self.product_price.text_content() or ""

    def is_add_to_cart_visible(self) -> bool:
        return self.add_to_cart_btn.is_visible()

    # ── Actions ─────────────────────────────────────────────────────
    def add_to_cart(self, quantity: int = 1) -> "ProductPage":
        if quantity > 1:
            self.quantity_input.fill(str(quantity))
        self.add_to_cart_btn.click()
        return self

    def wait_for_success_toast(self) -> "ProductPage":
        self.success_toast.wait_for(timeout=5_000)
        return self
```

```python
# tests/test_search.py
# Run with: pytest tests/test_search.py -v
import pytest
from playwright.sync_api import Page, expect

BASE = "https://practicesoftwaretesting.com"

def do_search(page: Page, term: str) -> None:
    """Helper: navigate to homepage, search, wait for results."""
    page.goto(BASE)
    page.wait_for_load_state("networkidle")
    page.get_by_role("textbox", name="Search").fill(term)
    page.get_by_role("textbox", name="Search").press("Enter")
    page.wait_for_load_state("networkidle")

# Parametrized: 4 search terms, all should return results
@pytest.mark.parametrize("term", [
    "Hammer",
    "Pliers",
    "Screwdriver",
    "Bolt Cutters",
])
def test_search_term_returns_results(page: Page, term: str):
    """Valid product names should return at least one result."""
    do_search(page, term)
    count = page.locator(".card").count()
    assert count > 0, f"Expected results for '{term}', found {count}"

def test_nonexistent_search_shows_no_results(page: Page):
    """Searching for a nonexistent product should show no results."""
    do_search(page, "XYZNONEXISTENTPRODUCT99999")
    # Either cards = 0 or a "no results" message appears
    card_count = page.locator(".card").count()
    no_results = page.locator("[class*='no-result'], .alert, p:has-text('no')").count()
    assert card_count == 0 or no_results > 0, \
        "Expected no results or a 'no results' message"
```

```python
# tests/test_auth.py
import pytest
from playwright.sync_api import Page, expect

BASE = "https://practicesoftwaretesting.com"
LOGIN_URL = f"{BASE}/auth/login"

VALID_USERS = [
    pytest.param(
        "customer@practicesoftwaretesting.com", "welcome01", "/account",
        id="customer"
    ),
    # Add more user roles if the site has them (admin, etc.)
]

INVALID_CREDS = [
    pytest.param("wrong@example.com",  "welcome01", id="wrong-email"),
    pytest.param("customer@practicesoftwaretesting.com", "wrongpass", id="wrong-password"),
    pytest.param("",                    "welcome01",   id="empty-email"),
]

@pytest.mark.parametrize("email,password,expected_path", VALID_USERS)
def test_valid_login(page: Page, email: str, password: str, expected_path: str):
    """Valid credentials should redirect to the expected path."""
    page.goto(LOGIN_URL)
    page.get_by_label("Email").fill(email)
    page.get_by_label("Password").fill(password)
    page.get_by_role("button", name="Login").click()
    page.wait_for_url(f"**{expected_path}")
    assert expected_path in page.url

@pytest.mark.parametrize("email,password", INVALID_CREDS)
def test_invalid_login_shows_error(page: Page, email: str, password: str):
    """Invalid credentials should keep user on login page or show error."""
    page.goto(LOGIN_URL)
    page.get_by_label("Email").fill(email)
    page.get_by_label("Password").fill(password)
    page.get_by_role("button", name="Login").click()
    error = page.locator(".alert-danger, .invalid-feedback, [role='alert']")
    assert "login" in page.url or error.is_visible(), \
        f"Expected error for email='{email}', password='{password}'"

def test_account_heading_visible_after_login(authenticated_page: Page):
    """After login, account heading should be visible."""
    heading = authenticated_page.get_by_role("heading", name="My account")
    expect(heading).to_be_visible()

def test_sign_out_returns_to_home(authenticated_page: Page):
    """Signing out should navigate away from /account."""
    sign_out = authenticated_page.get_by_role("link", name="Sign out")
    sign_out.click()
    assert "/account" not in authenticated_page.url
```
