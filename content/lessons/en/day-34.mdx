## Concept

<En>
**Final project structure:**
```
playwright-todo-suite/
├── .github/
│   └── workflows/
│       └── playwright.yml
├── pages/
│   ├── __init__.py
│   ├── base_page.py
│   └── todo_page.py
├── tests/
│   ├── __init__.py
│   ├── conftest.py
│   ├── test_add_items.py
│   ├── test_complete_items.py
│   ├── test_filters.py
│   └── test_edge_cases.py
├── screenshots/         # gitignored
├── pytest.ini
├── requirements.txt
└── README.md
```

**What "production quality" means here:**
- No raw locators in test files — everything goes through the page object
- Every test has a clear, descriptive name
- Shared setup lives in fixtures, not duplicated in tests
- Edge cases are covered (empty state, special characters, very long text)
- CI is configured to run on every push
- Screenshots are captured on failure
- The suite runs in under 60 seconds
</En>
<Vi>

**"Chất lượng production" có nghĩa là:**
- Không có locator thô trong file test — mọi thứ đi qua page object
- Mỗi test có tên rõ ràng, mô tả
- Setup dùng chung nằm trong fixture, không trùng lặp trong test
- Các edge case được bao phủ
- CI được cấu hình để chạy trên mỗi lần push
</Vi>

# tests/conftest.py — shared fixtures
import pytest
from pathlib import Path
from playwright.sync_api import Page, BrowserContext
from pages.todo_page import TodoPage

SCREENSHOTS_DIR = Path("screenshots")

@pytest.fixture(autouse=True)
def screenshot_on_failure(page: Page, request):
    """Auto-capture screenshot on any test failure."""
    yield
    if request.node.rep_call.failed:
        SCREENSHOTS_DIR.mkdir(exist_ok=True)
        page.screenshot(path=str(SCREENSHOTS_DIR / f"{request.node.name}.png"))

@pytest.fixture
def todo(page: Page) -> TodoPage:
    tp = TodoPage(page)
    tp.navigate()
    return tp

@pytest.fixture
def todo_with_three_items(todo: TodoPage) -> TodoPage:
    for text in ["First item", "Second item", "Third item"]:
        todo.add(text)
    return todo
```

# tests/test_add_items.py
import pytest
from playwright.sync_api import expect
from pages.todo_page import TodoPage

def test_empty_state_on_load(todo: TodoPage):
    assert todo.get_count() == 0

@pytest.mark.parametrize("text", [
    "Normal todo item",
    "Item with numbers 12345",
    "Item with special chars !@#$%",
    "A very long todo item that exceeds normal length expectations by quite a bit",
])
def test_add_various_items(todo: TodoPage, text: str):
    todo.add(text)
    todo.expect_item(text)
    assert todo.get_count() == 1

def test_add_multiple_items_in_sequence(todo: TodoPage):
    items = ["Item A", "Item B", "Item C"]
    for item in items:
        todo.add(item)
    assert todo.get_count() == len(items)
    for item in items:
        todo.expect_item(item)

def test_empty_input_does_not_add_item(todo: TodoPage):
    todo.new_todo_input.press("Enter")
    assert todo.get_count() == 0
```

```python
# tests/test_complete_items.py
from pages.todo_page import TodoPage

def test_mark_single_item_complete(todo_with_three_items: TodoPage):
    todo_with_three_items.complete("First item")
    completed = todo_with_three_items.page.locator(".todo-list li.completed")
    assert completed.count() == 1

def test_mark_all_complete(todo_with_three_items: TodoPage):
    for text in ["First item", "Second item", "Third item"]:
        todo_with_three_items.complete(text)
    completed = todo_with_three_items.page.locator(".todo-list li.completed")
    assert completed.count() == 3

def test_delete_item(todo_with_three_items: TodoPage):
    initial_count = todo_with_three_items.get_count()
    todo_with_three_items.delete("Second item")
    todo_with_three_items.expect_item_absent("Second item")
    assert todo_with_three_items.get_count() == initial_count - 1
```
