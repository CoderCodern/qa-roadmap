## Intro

<En>
Testing one input is good. Testing ten inputs with one test function is powerful. **Data-driven testing** means running the same test logic with multiple sets of inputs — valid credentials, invalid credentials, edge cases, boundary values — all from a single test definition.

pytest's `@pytest.mark.parametrize` decorator makes this clean and easy. Instead of copying a test five times with different values, you write it once and provide a list of inputs. Each combination becomes its own test case with its own pass/fail result in the report.

Data-driven testing is how you achieve thorough coverage without code duplication.
</En>
<Vi>
Kiểm thử một đầu vào là tốt. Kiểm thử mười đầu vào với một hàm test là mạnh mẽ. **Kiểm thử theo dữ liệu** có nghĩa là chạy cùng logic test với nhiều bộ đầu vào — thông tin đăng nhập hợp lệ, không hợp lệ, trường hợp biên, giá trị ranh giới — tất cả từ một định nghĩa test duy nhất.

Decorator `@pytest.mark.parametrize` của pytest làm cho điều này sạch sẽ và dễ dàng. Thay vì sao chép test năm lần với các giá trị khác nhau, bạn viết một lần và cung cấp danh sách đầu vào. Mỗi tổ hợp trở thành một test case riêng với kết quả pass/fail riêng trong báo cáo.
</Vi>

## Concept

<En>
**Basic syntax — single parameter:**

```python
import pytest

@pytest.mark.parametrize("search_term", [
    "Hammer",
    "Pliers",
    "Screwdriver",
    "Drill",
])
def test_search_returns_results(page, search_term):
    page.goto("https://practicesoftwaretesting.com")
    page.get_by_role("textbox", name="Search").fill(search_term)
    page.get_by_role("textbox", name="Search").press("Enter")
    page.wait_for_load_state("networkidle")
    assert page.locator(".card").count() > 0
# This generates 4 separate test cases:
# test_search_returns_results[Hammer]
# test_search_returns_results[Pliers]
# test_search_returns_results[Screwdriver]
# test_search_returns_results[Drill]
```

**Multiple parameters — tuples:**

```python
@pytest.mark.parametrize("email,password,expected_url", [
    # (email, password, expected_url_fragment)
    ("customer@practicesoftwaretesting.com", "welcome01", "/account"),   # valid
    ("admin@practicesoftwaretesting.com",    "AKZo8tZ6",  "/admin"),     # admin
])
def test_login_redirects(page, email, password, expected_url):
    page.goto("https://practicesoftwaretesting.com/auth/login")
    page.get_by_label("Email").fill(email)
    page.get_by_label("Password").fill(password)
    page.get_by_role("button", name="Login").click()
    assert expected_url in page.url
```

**Naming parameters with `pytest.param` and `ids`:**

```python
@pytest.mark.parametrize("email,password", [
    pytest.param("customer@practicesoftwaretesting.com", "welcome01",
                 id="valid-customer"),
    pytest.param("invalid@example.com", "wrongpassword",
                 id="invalid-credentials"),
    pytest.param("", "welcome01",
                 id="empty-email"),
], )
def test_login_cases(page, email, password):
    # ...
```

**Expected failures — `pytest.param` with `marks`:**

```python
@pytest.mark.parametrize("search_term,min_results", [
    ("Hammer", 1),
    ("Pliers", 1),
    pytest.param("XYZ_NONEXISTENT_PRODUCT_999", 0, id="no-results"),
])
def test_search(page, search_term, min_results):
    # ...
    assert page.locator(".card").count() >= min_results
```

**Combining parametrize with fixtures:**

```python
@pytest.mark.parametrize("product_name", ["Bolt Cutters", "Pliers", "Hammer"])
def test_product_appears_in_search(page, product_name):
    """Parametrize works with pytest fixtures — just declare both."""
    page.goto("https://practicesoftwaretesting.com")
    page.get_by_role("textbox", name="Search").fill(product_name)
    page.get_by_role("textbox", name="Search").press("Enter")
    page.wait_for_load_state("networkidle")
    expect(page.locator(".card")).not_to_have_count(0)
```

**Loading data from a list or file:**

```python

```

import pytest
from playwright.sync_api import Page, expect

BASE = "https://practicesoftwaretesting.com"

@pytest.mark.parametrize("search_term", [
    "Hammer",
    "Pliers",
    "Bolt Cutters",
    "Screwdriver",
])
def test_search_term_returns_results(page: Page, search_term: str):
    """Each search term should return at least one product."""
    page.goto(BASE)
    search = page.get_by_role("textbox", name="Search")
    search.fill(search_term)
    search.press("Enter")
    page.wait_for_load_state("networkidle")

    cards = page.locator(".card")
    assert cards.count() > 0, f"No results found for '{search_term}'"

# Run output:
# test_parametrize_search.py::test_search_term_returns_results[Hammer]       PASSED

# tests/test_parametrize_login.py
# Run with: pytest tests/test_parametrize_login.py -v
import pytest
from playwright.sync_api import Page, expect

BASE = "https://practicesoftwaretesting.com"
LOGIN_URL = f"{BASE}/auth/login"

# Test data: (description_id, email, password, should_succeed, expected_url_fragment)
LOGIN_CASES = [
    pytest.param(
        "customer@practicesoftwaretesting.com", "welcome01", True, "/account",
        id="valid-customer-login"
    ),
    pytest.param(
        "wrong@example.com", "welcome01", False, "/auth/login",
        id="wrong-email"
    ),
    pytest.param(
        "customer@practicesoftwaretesting.com", "wrongpassword", False, "/auth/login",
        id="wrong-password"
    ),
    pytest.param(
        "", "welcome01", False, "/auth/login",
        id="empty-email"
    ),
]

@pytest.mark.parametrize("email,password,should_succeed,expected_url", LOGIN_CASES)
def test_login_scenarios(page: Page, email: str, password: str,
                         should_succeed: bool, expected_url: str):
    """Parametrized login: tests both valid and invalid credential combos."""
    page.goto(LOGIN_URL)
    page.get_by_label("Email").fill(email)
    page.get_by_label("Password").fill(password)
    page.get_by_role("button", name="Login").click()

    if should_succeed:
        page.wait_for_url("**/account")
        assert expected_url in page.url, f"Expected {expected_url}, got {page.url}"
    else:
        # Stay on login or show error
        error = page.locator(".alert-danger")
        assert expected_url in page.url or error.is_visible(), \
            f"Expected error or stay on login, got URL: {page.url}"
```

# Demonstrates parametrize + fixtures working together
import pytest
from playwright.sync_api import Page

BASE = "https://practicesoftwaretesting.com"

@pytest.fixture
def homepage(page: Page):
    page.goto(BASE)
    page.wait_for_load_state("networkidle")
    return page

@pytest.mark.parametrize("category,min_count", [
    ("Hammer", 1),
    ("Pliers", 1),
    ("Wrench", 1),
])
def test_category_search_has_results(homepage: Page, category: str, min_count: int):
    """Uses homepage fixture AND parametrize — they compose cleanly."""
    homepage.get_by_role("textbox", name="Search").fill(category)
    homepage.get_by_role("textbox", name="Search").press("Enter")
    homepage.wait_for_load_state("networkidle")

    count = homepage.locator(".card").count()
    assert count >= min_count, f"Expected >= {min_count} results for '{category}', got {count}"
```
