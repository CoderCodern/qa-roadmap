## Intro

<En>
Yesterday you learned the theory of Page Object Model. Today you build one from scratch — step by step — for a real site. By the end you will have a working `LoginPage` and `AccountPage` class with tests that use them.

The target site is `https://practicesoftwaretesting.com`. We will:
1. Inspect the login form to identify selectors
2. Build `LoginPage` with navigate, fill, and submit actions
3. Build `AccountPage` to verify the logged-in state
4. Write tests that wire them together
5. Add a shared `logged_in` pytest fixture so any test can skip login

## Concept

<En>
**Project structure we are building:**

```
project/
├── pages/
│   ├── __init__.py       ← makes pages/ a Python package
│   ├── login_page.py     ← LoginPage class
│   └── account_page.py   ← AccountPage class
├── tests/
│   ├── conftest.py       ← shared fixtures (logged_in, base_url)
│   └── test_auth.py      ← test file using both page objects
└── pytest.ini            ← base_url and other config
```

**Key design decisions:**
- Locators are `Locator` typed attributes set in `__init__`
- Action methods return the next logical Page Object after navigation
- `conftest.py` defines a `logged_in_page` fixture that tests can declare as a parameter to get a pre-authenticated page
- Tests import page objects using relative imports from the `pages/` package

**The two Page Objects we build:**

```
LoginPage                       AccountPage
──────────────────────────      ──────────────────────────
+ URL                           + URL
+ email_input: Locator          + welcome_heading: Locator
+ password_input: Locator       + nav_menu: Locator
+ login_button: Locator
+ error_alert: Locator
──────────────────────────      ──────────────────────────
+ navigate() → LoginPage        + is_logged_in() → bool
+ login(email, pw) → AccountPage
+ get_error_text() → str
```
</En>

```python
# pages/login_page.py
from __future__ import annotations
from playwright.sync_api import Page, Locator

class LoginPage:
    """
    Page Object for: https://practicesoftwaretesting.com/auth/login

    Usage:
        login_page = LoginPage(page)
        login_page.navigate()
        account_page = login_page.login("email", "password")
    """

    URL = "https://practicesoftwaretesting.com/auth/login"

    def __init__(self, page: Page) -> None:
        self.page = page

        # ── Locators ────────────────────────────────────────────────
        self.email_input: Locator = page.get_by_label("Email")
        self.password_input: Locator = page.get_by_label("Password")
        self.login_button: Locator = page.get_by_role("button", name="Login")
        self.error_alert: Locator = page.locator(".alert-danger")
        self.register_link: Locator = page.get_by_role("link", name="Register your account")

    # ── Navigation ──────────────────────────────────────────────────
    def navigate(self) -> LoginPage:
        """Go to the login page."""
        self.page.goto(self.URL)
        return self

    # ── Actions ─────────────────────────────────────────────────────
    def fill_email(self, email: str) -> LoginPage:
        self.email_input.fill(email)
        return self

    def fill_password(self, password: str) -> LoginPage:
        self.password_input.fill(password)
        return self

    def submit(self) -> LoginPage:
        self.login_button.click()
        return self

    def login(self, email: str, password: str) -> "AccountPage":
        """
        Fill credentials and click Login.
        Returns AccountPage (assuming login succeeds).
        For negative tests, check error_alert after calling this.
        """
        from pages.account_page import AccountPage
        self.fill_email(email)
        self.fill_password(password)
        self.submit()
        return AccountPage(self.page)

    # ── Queries ─────────────────────────────────────────────────────
    def get_error_text(self) -> str:
        """Return the error message text, or empty string if not visible."""
        if self.error_alert.is_visible():
            return self.error_alert.text_content() or ""
        return ""

    def is_error_visible(self) -> bool:
        return self.error_alert.is_visible()
```

```python
# pages/account_page.py
from __future__ import annotations
from playwright.sync_api import Page, Locator

class AccountPage:
    """
    Page Object for: https://practicesoftwaretesting.com/account

    Represents the logged-in account/dashboard page.
    """

    URL = "https://practicesoftwaretesting.com/account"

    def __init__(self, page: Page) -> None:
        self.page = page

        # ── Locators ────────────────────────────────────────────────
        self.welcome_heading: Locator = page.get_by_role("heading", name="My account")
        self.sign_out_link: Locator = page.get_by_role("link", name="Sign out")
        self.profile_link: Locator = page.get_by_role("link", name="Profile")
        self.orders_link: Locator = page.get_by_role("link", name="My orders")

    # ── Queries ─────────────────────────────────────────────────────
    def is_logged_in(self) -> bool:
        """Return True if the account page heading is visible."""
        return self.welcome_heading.is_visible()

    def wait_until_loaded(self) -> "AccountPage":
        """Wait for the account page to fully load."""
        self.page.wait_for_url("**/account", timeout=10_000)
        self.welcome_heading.wait_for()
        return self

    # ── Actions ─────────────────────────────────────────────────────
    def sign_out(self) -> None:
        """Click Sign out."""
        self.sign_out_link.click()
```

```python
# tests/conftest.py
import sys, os
sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))

import pytest
from playwright.sync_api import Page
from pages.login_page import LoginPage

VALID_EMAIL = "customer@practicesoftwaretesting.com"
VALID_PASSWORD = "welcome01"

@pytest.fixture
def logged_in_page(page: Page) -> Page:
    """
    Fixture: returns a Page that is already logged in.
    Use this in tests that need to skip the login step.

    Usage in test:
        def test_something(logged_in_page: Page):
            # page is already at /account
            expect(logged_in_page).to_have_url("**/account")
    """
    login_page = LoginPage(page)
    login_page.navigate()
    login_page.login(VALID_EMAIL, VALID_PASSWORD)
    page.wait_for_url("**/account")
    return page
```

```python
# tests/test_auth.py
# Run with: pytest tests/test_auth.py -v
import pytest
from playwright.sync_api import Page, expect
from pages.login_page import LoginPage
from pages.account_page import AccountPage

BASE = "https://practicesoftwaretesting.com"
VALID_EMAIL = "customer@practicesoftwaretesting.com"
VALID_PASSWORD = "welcome01"

class TestLogin:
    """Tests for the login flow."""

    def test_valid_login_redirects_to_account(self, page: Page):
        """Successful login should redirect to /account."""
        account = LoginPage(page).navigate().login(VALID_EMAIL, VALID_PASSWORD)
        account.wait_until_loaded()
        assert account.is_logged_in(), "Account heading not visible"
        expect(page).to_have_url(f"{BASE}/account")

    def test_invalid_password_shows_error(self, page: Page):
        """Wrong password should show an error alert."""
        login_page = LoginPage(page)
        login_page.navigate()
        login_page.fill_email(VALID_EMAIL)
        login_page.fill_password("wrongpassword")
        login_page.submit()
        expect(login_page.error_alert).to_be_visible()
        assert "Invalid" in login_page.get_error_text() or login_page.is_error_visible()

    def test_empty_fields_do_not_navigate(self, page: Page):
        """Submitting empty form should not navigate away from login."""
        login_page = LoginPage(page)
        login_page.navigate()
        login_page.submit()
        assert "login" in page.url

class TestAccountPage:
    """Tests that use the logged_in_page fixture."""

    def test_account_heading_visible(self, logged_in_page: Page):
        """Account page should show the welcome heading."""
        account = AccountPage(logged_in_page)
        expect(account.welcome_heading).to_be_visible()

    def test_sign_out_link_visible(self, logged_in_page: Page):
        """Sign out link should be visible when logged in."""
        account = AccountPage(logged_in_page)
        expect(account.sign_out_link).to_be_visible()
```
